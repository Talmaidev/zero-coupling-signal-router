+/* zcsr culling — viewport culling for 2D AABBs. Pure logic, no GL, no heap.
+ *
+ * Two strategies:
+ *   - linear: test every box against the view (O(n), cache-friendly).
+ *   - uniform grid: hash boxes into fixed-size cells, then walk only the cells the
+ *     view overlaps (fast when the view covers a small fraction of the world).
+ *
+ * The caller owns all memory: the box array, the grid scratch, and the output index list.
+ */
+#ifndef ZCSR_CULLING_H
+#define ZCSR_CULLING_H
+
+#include <stdint.h>
+#include <stddef.h>
+
+#ifdef __cplusplus
+extern "C" {
+#endif
+
+/* Axis-aligned box in world/pixel coordinates. */
+typedef struct {
+    float min_x, min_y;
+    float max_x, max_y;
+} zcsr_aabb_t;
+
+/* Visible region (camera/viewport) in the same coordinate space. */
+typedef struct {
+    float min_x, min_y;
+    float max_x, max_y;
+} zcsr_view_t;
+
+/* Returns 1 if the box intersects the view (inclusive edges), else 0. */
+int zcsr_aabb_in_view(const zcsr_aabb_t* box, const zcsr_view_t* view);
+
+/* Linear cull: test every box. Writes the indices of visible boxes into out_indices and
+ * returns the count. Stops early if out_capacity is reached. */
+size_t zcsr_cull_linear(const zcsr_aabb_t* boxes, size_t count,
+                        const zcsr_view_t* view,
+                        uint32_t* out_indices, size_t out_capacity);
+
+/* ---- uniform grid ---- */
+typedef struct {
+    float origin_x, origin_y;   /* world coordinate of cell (0,0) corner */
+    float cell_w, cell_h;       /* cell size (pixels) */
+    uint32_t cols, rows;        /* grid dimensions */
+    /* Packed linked-list storage. cell_head[c] is the first box index in cell c, or
+     * ZCSR_CULL_EMPTY; next[b] is the next box in that cell, or ZCSR_CULL_EMPTY. */
+    uint32_t* cell_head;        /* length cols*rows */
+    uint32_t* next;             /* length box_count */
+} zcsr_grid_t;
+
+#define ZCSR_CULL_EMPTY 0xFFFFFFFFu
+
+/* Build a grid: clear the heads, then insert each box into every cell it overlaps.
+ * Returns the number of (box, cell) insertions performed. A box larger than a cell is
+ * inserted into multiple cells. The caller must size cell_head and next. */
+size_t zcsr_grid_build(zcsr_grid_t* grid, const zcsr_aabb_t* boxes, size_t count);
+
+/* Query the grid for boxes overlapping the view. Writes de-duplicated visible box indices
+ * into out_indices (the 'seen' array, length box_count, is used to dedupe and is cleared
+ * internally). A box is returned only if its AABB truly intersects the view, since a large
+ * box can be linked from a cell just outside the view. Returns the visible count. */
+size_t zcsr_grid_query(const zcsr_grid_t* grid, const zcsr_aabb_t* boxes, size_t box_count,
+                       const zcsr_view_t* view,
+                       uint32_t* out_indices, size_t out_capacity,
+                       uint8_t* seen);
+
+/* Fraction of the world covered by the view (0..1), to choose a strategy. */
+float zcsr_view_coverage(const zcsr_grid_t* grid, const zcsr_view_t* view);
+
+#ifdef __cplusplus
+}
+#endif
+
+#endif /* ZCSR_CULLING_H */

+/* Culling bench — linear vs uniform grid across view sizes.
+ *
+ * 10000 boxes spread over a 4000x4000 world. The view shrinks from covering the whole world

+ */

+#define WORLD    4000
+#define CELL     100

+    static LARGE_INTEGER freq;
+    LARGE_INTEGER ctr;
+    if (freq.QuadPart == 0) QueryPerformanceFrequency(&freq);
+    QueryPerformanceCounter(&ctr);
+    return (uint64_t)((double)ctr.QuadPart * 1e9 / (double)freq.QuadPart);
+}
+#else
+#  include <time.h>
+static uint64_t now_ns(void) {
+    struct timespec ts;
+    clock_gettime(CLOCK_MONOTONIC, &ts);

+}

+static double bench_linear(const zcsr_aabb_t* boxes, const zcsr_view_t* view,
+                           uint32_t* out, size_t* visible) {

+    for (int i = 0; i < ITERS; ++i) {
+        uint64_t t0 = now_ns();
+        *visible = zcsr_cull_linear(boxes, NBOX, view, out, NBOX);

+}

+int zcsr_aabb_in_view(const zcsr_aabb_t* b, const zcsr_view_t* v) {
+    if (!b || !v) return 0;
+    return b->min_x <= v->max_x && b->max_x >= v->min_x &&
+           b->min_y <= v->max_y && b->max_y >= v->min_y;

+size_t zcsr_cull_linear(const zcsr_aabb_t* boxes, size_t count,
+                        const zcsr_view_t* view,
+                        uint32_t* out_indices, size_t out_capacity) {
+    size_t i, n = 0;
+    if (!boxes || !view || !out_indices) return 0;
+    for (i = 0; i < count; ++i) {
+        if (zcsr_aabb_in_view(&boxes[i], view)) {
+            if (n >= out_capacity) return n;
+            out_indices[n++] = (uint32_t)i;
+        }
+    }
+    return n;
+}

+static uint32_t zcsr_clamp_i(int v, int lo, int hi) {
+    if (v < lo) return (uint32_t)lo;
+    if (v > hi) return (uint32_t)hi;
+    return (uint32_t)v;
+}

+size_t zcsr_grid_build(zcsr_grid_t* g, const zcsr_aabb_t* boxes, size_t count) {
+    uint32_t cells, c, r;
+    size_t i, inserts = 0;
+    if (!g || !boxes || !g->cell_head || !g->next) return 0;
+    cells = g->cols * g->rows;
+    for (c = 0; c < cells; ++c) g->cell_head[c] = ZCSR_CULL_EMPTY;

+        const zcsr_aabb_t* b = &boxes[i];
+        int c0 = (int)((b->min_x - g->origin_x) / g->cell_w);

+/* LSD radix sort of instances by tex (uint32), 4 passes of 8 bits. Scratch must hold
+ * count instances. Stable, no heap. */
+typedef uint32_t (*zcsr_instance_key_fn)(const zcsr_gl_instance*);

+static void zcsr_gl_radix(zcsr_gl_instance* items, size_t count,
+                          zcsr_gl_instance* scratch, zcsr_instance_key_fn key_of) {
+    int shift, b;
+    for (shift = 0; shift < 32; shift += 8) {
+        uint32_t hist[256];
+        size_t i;
+        for (b = 0; b < 256; ++b) hist[b] = 0;
+        for (i = 0; i < count; ++i) ++hist[(key_of(&items[i]) >> shift) & 0xFFu];
+        uint32_t total = 0;
+        for (b = 0; b < 256; ++b) {
+            uint32_t c = hist[b];
+            hist[b] = total;
+            total += c;
+        }
+        for (i = 0; i < count; ++i) {
+            uint32_t key = key_of(&items[i]);
+            scratch[hist[(key >> shift) & 0xFFu]++] = items[i];
+        }
+        memcpy(items, scratch, count * sizeof(zcsr_gl_instance));
+    }
+}

+/* State sort unit tests — opaque/translucent partition, opaque texture order, translucent
+ * back-to-front z order. Pure logic, headless. Uses CHECK (runs in Release). */
+#include "zcsr/gl_render.h"
+
+#include <stdio.h>
+#include <string.h>
+
+#define N 20
+
+#define CHECK(cond) do { if (!(cond)) { printf("FAIL line %d: %s\n", __LINE__, #cond); return 1; } } while (0)
+
+static zcsr_gl_instance make(uint32_t tex, float a, float z, float mode) {
+    zcsr_gl_instance in;
+    memset(&in, 0, sizeof in);
+    in.cx = 0; in.cy = 0; in.w = 8; in.h = 8;
+    in.u0 = 0; in.v0 = 0; in.u1 = 1; in.v1 = 1;
+    in.r = 1; in.g = 1; in.b = 1; in.a = a;
+    in.mode = mode; in.tex = tex; in.z = z;
+    return in;
+}
+
+int main(void) {
+    static zcsr_gl_instance items[N];
+    static zcsr_gl_instance scratch[N];
+    size_t opaque;
+
+    /* 10 opaque (a=1, mode 0), 10 translucent (a=0.5), interleaved. */
+    for (int i = 0; i < N; ++i) {
+        int translucent = (i % 2) == 1;
+        uint32_t tex = (uint32_t)(10 - i / 2); /* descending texture ids */
+        float z = (float)i;                     /* increasing z */
+        items[i] = make(tex, translucent ? 0.5f : 1.0f, z, 0.0f);
+    }
+
+    zcsr_sort_state(items, N, scratch, &opaque);
+    CHECK(opaque == 10);
+
+    /* first 10 are opaque, sorted by tex ascending */
+    for (size_t i = 0; i < opaque; ++i) {
+        CHECK(items[i].a == 1.0f);
+        if (i > 0) CHECK(items[i].tex >= items[i - 1].tex);
+    }
+
+    /* last 10 are translucent, sorted by z descending (far first) */
+    for (size_t i = opaque; i < N; ++i) {
+        CHECK(items[i].a == 0.5f);
+        if (i > opaque) CHECK(items[i].z <= items[i - 1].z);
+    }
+
+    /* MIX/ADD with a=1 is still translucent */
+    {
+        zcsr_gl_instance mix[2];
+        mix[0] = make(1, 1.0f, 0.0f, 1.0f); /* MIX -> translucent */
+        mix[1] = make(2, 1.0f, 0.0f, 0.0f); /* opaque */
+        size_t op;
+        zcsr_sort_state(mix, 2, scratch, &op);
+        CHECK(op == 1);
+        CHECK(mix[0].tex == 2); /* opaque one first */
+    }
+
+    printf("test_state: PASS (opaque=%zu, texture-asc / z-desc)\n", opaque);
+    return 0;
+}
